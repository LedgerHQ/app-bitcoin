# Bitcoin-specific symbols to zero out (in addition to framework SDK list).
#
# These are app-level globals and mock state that are irrelevant to
# Bitcoin's core protocol fuzzing and would bloat the Absolution prefix.
# Zeroing removes a symbol from the prefix entirely (zero-cost fixed 0).

G_dispatcher_context
g_ui_state
G_sign_psbt_cache
pairs

# ── Pointers overwritten by harness every iteration ──────────────────
# Random prefix bytes in pointer slots crash; harness sets correct values.
G_processing_screen_text
G_swap_sign_return_value_address@handle_swap_sign_transaction.c

# ── SDK swap return-value pointer (lib_standard_app/swap_utils.h) ────
# Only dereferenced under #ifdef HAVE_SWAP in io.c, which is NOT compiled
# in the bitcoin fuzz build.  Zero it so the 8-byte pointer slot is
# eliminated from the prefix (zero-cost) and any future swap enablement
# starts from a defined NULL state.
G_swap_signing_return_value_address@lib_standard_app

# ── Timeout bookkeeping (io_ext.c) ───────────────────────────────────
# G_ticks and the two timeout-start ticks are set by the SDK ticker
# callback/timeout-start helpers.  None of them gates parser branches
# (only app_ticker_event_callback reads them, and that path is bounded
# by G_is_timeout_active which starts false).  Zero them so ~6 prefix
# bytes are reclaimed and spurious EXCEPTION_IO_RESET paths never fire.
G_ticks@io_ext.c
G_interruption_timeout_start_tick@io_ext.c
G_processing_timeout_start_tick@io_ext.c
G_is_timeout_active@io_ext.c

# ── Stack canary (constant, never mutated) ───────────────────────────
app_stack_canary

# ── Crypto context (920B internal to mock, no app branches) ─────────
G_cx
